cmake_minimum_required(VERSION 3.20)
project(SOLNPInstallationTest LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
get_filename_component(SOLNP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../source" ABSOLUTE)
find_package(LAPACK REQUIRED)
find_library(OSQP_LIBRARY NAMES osqp HINTS "$ENV{OSQP_HOME}/lib" REQUIRED NO_DEFAULT_PATH)
find_path(OSQP_INCLUDE_DIR osqp.h HINTS "$ENV{OSQP_HOME}/include/osqp" "$ENV{OSQP_HOME}/include" REQUIRED NO_DEFAULT_PATH)
set(NAMES der_info linalg linsys qp_solver Rescue solnp_py solnp subnp_qp solnp_util solnp_c_interface)
foreach(N IN LISTS NAMES)
    list(APPEND SOURCES "${SOLNP_ROOT}/source/${N}.c")
endforeach()
add_library(solnp SHARED ${SOURCES})
target_include_directories(solnp PUBLIC "${SOLNP_ROOT}/include" "${OSQP_INCLUDE_DIR}")
target_compile_definitions(solnp PRIVATE ADD_)
target_link_libraries(solnp PUBLIC ${OSQP_LIBRARY} ${LAPACK_LIBRARIES} m)
add_executable(test_nlp test_nlp.c)
target_link_libraries(test_nlp PRIVATE solnp)
enable_testing()
add_test(NAME coinor_nlp_install COMMAND test_nlp ${SOLNP_ROOT}/../examples/problem.txt)
install(TARGETS solnp LIBRARY DESTINATION lib)
install(DIRECTORY "${SOLNP_ROOT}/include/" DESTINATION include/solnp FILES_MATCHING PATTERN "*.h")

# Keep installed library usable with the configured external OSQP/BLAS.
set_target_properties(solnp PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
